Private Sub Worksheet_FollowHyperlink(ByVal Target As Hyperlink)
    Dim parts() As String, entityID As String, entityType As String
    parts = Split(Target.TextToDisplay, " - ")
    If UBound(parts) >= 1 Then
        entityID = parts(0)
        entityType = parts(1)
        Select Case LCase(entityType)
            Case "account": ShowAccountDetails entityID
            Case "master account": ShowMasterDetails entityID
            Case "relationship": ShowRelationshipDetails entityID
            Case Else: MsgBox "Unknown entity type."
        End Select
    End If
End Sub
