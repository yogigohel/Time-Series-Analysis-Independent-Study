Option Explicit

Private filePath As String

'-----------------------------------------------------------------------------
' UserForm Initialize: just clear the label, uncheck overwrites, clear listbox
'-----------------------------------------------------------------------------
Private Sub UserForm_Initialize()
    Me.lblFilePath.Caption = ""
    Me.chkAllowOverwrites.Value = False
    Me.lstPreview.Clear
    Me.lstPreview.ColumnCount = 1
    Me.lstPreview.ColumnWidths = "400"
    filePath = ""
End Sub

'-----------------------------------------------------------------------------
' BROWSE: Let the user pick an external file
'-----------------------------------------------------------------------------
Private Sub cmdBrowse_Click()
    On Error GoTo ErrHandler
    Dim fd As FileDialog
    Set fd = Application.FileDialog(msoFileDialogFilePicker)
    
    With fd
        .Title = "Select Accounts File"
        .Filters.Clear
        .Filters.Add "Excel Files", "*.xlsx; *.xlsm; *.xls; *.csv", 1
        .AllowMultiSelect = False
        
        If .Show = -1 Then
            filePath = .SelectedItems(1)
            Me.lblFilePath.Caption = filePath
            
            ' Optionally, show a partial preview:
            Me.lstPreview.Clear
            Me.lstPreview.AddItem "Selected file: " & filePath
            Me.lstPreview.AddItem "Ready to import..."
        End If
    End With
CleanExit:
    Exit Sub
ErrHandler:
    MsgBox "Error in cmdBrowse_Click: " & Err.Description, vbCritical
    Resume CleanExit
End Sub

'-----------------------------------------------------------------------------
' IMPORT: calls modUpload.ImportAccounts
'-----------------------------------------------------------------------------
Private Sub cmdImport_Click()
    On Error GoTo ErrHandler
    
    If filePath = "" Then
        MsgBox "No file selected. Please click Browse first.", vbExclamation
        Exit Sub
    End If
    
    ' If user wants overwrites (CHANGED items => update "Account Data"), pass True
    Dim allowDup As Boolean
    allowDup = Me.chkAllowOverwrites.Value
    
    ' Call the module function
    modUpload.ImportAccounts filePath, allowDup
    
    MsgBox "Data imported successfully.", vbInformation
    Unload Me
CleanExit:
    Exit Sub
ErrHandler:
    MsgBox "Error in cmdImport_Click: " & Err.Description, vbCritical
    Resume CleanExit
End Sub
