Sub ImportAccounts()
    On Error GoTo ErrorHandler
    Application.ScreenUpdating = False
    
    Dim conn As WorkbookConnection
    For Each conn In ThisWorkbook.Connections
        If conn.Name = "Query - ImportQuery" Then
            conn.Refresh
        End If
    Next conn
    
    'Remove any potential duplicates
    ThisWorkbook.Sheets("Accounts").ListObjects("tblAccounts").RemoveDuplicates _
        Columns:=Array("Account#"), Header:=xlYes
    
    Application.ScreenUpdating = True
    MsgBox "Accounts updated successfully!", vbInformation
    Exit Sub

ErrorHandler:
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Application.ScreenUpdating = True
End Sub
