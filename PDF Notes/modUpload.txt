Option Explicit

' =================================================================================
' modUpload: A SINGLE module that handles all logic for:
'   1) Reading "Account Data" from this workbook
'   2) Reading external file columns by name (Relationship#, Master Account ID, etc.)
'   3) Comparing both sets to classify as NEW, CHANGED, REMOVED, or UNCHANGED
'   4) Importing NEW or CHANGED data into "Account Data" with highlighting
'   5) Logging changes in "Change Log"
'
' This approach uses only LATE BINDING to Scripting.Dictionary (CreateObject)
' => No special references or .Contains calls.
' =================================================================================

' -------------------------------
' Constants / Column Identifiers
' -------------------------------
Private Const SH_ACCOUNT_DATA As String = "Account Data"
Private Const SH_CHANGE_LOG   As String = "Change Log"

' "Account Data" columns (assuming row 1 has headers)
Private Enum AcDataCols
    cRelNum = 1         ' A -> Relationship#
    cRelName            ' B -> Relationship Name
    cMasterID           ' C -> Master Account ID
    cMasterName         ' D -> Master Account Name
    cAcctNum            ' E -> Account#
    cAcctName           ' F -> Acct Name
End Enum

' "Change Log" columns
Private Enum ChgCols
    chgTimestamp = 1
    chgUser
    chgAction
    chgAffectedData
    chgOldValue
    chgNewValue
End Enum

' Required header names in the external file
Private requiredHeaders() As Variant

' Set up once:
Private Sub SetupRequiredHeaders()
    ' This array must match your "Account Data" columns
    ' EXACT strings in row 1 of the external file
    requiredHeaders = Array("Relationship#", "Relationship Name", _
                            "Master Account ID", "Master Account Name", _
                            "Account#", "Acct Name")
End Sub

' -------------------------------------------------
' PUBLIC: Called from the userform's "Import" button
' -------------------------------------------------
Public Sub ImportAccounts(ByVal sourceFilePath As String, _
                          ByVal allowOverwrites As Boolean)
    ' This routine re-reads the external file, compares to "Account Data,"
    ' and updates the sheet with NEW (green) or CHANGED (yellow).
    
    On Error GoTo ErrHandler
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    
    SetupRequiredHeaders
    
    ' 1) Parse "Account Data" existing
    Dim dictExisting As Object
    Set dictExisting = CreateObject("Scripting.Dictionary")
    dictExisting.CompareMode = vbTextCompare
    BuildDictFromAccountData dictExisting
    
    ' 2) Parse external file
    Dim dictUpload As Object
    Set dictUpload = CreateObject("Scripting.Dictionary")
    dictUpload.CompareMode = vbTextCompare
    BuildDictFromExternalFile sourceFilePath, dictUpload
    
    ' 3) For each account in dictUpload, decide if NEW or CHANGED
    Dim key As Variant
    For Each key In dictUpload.Keys
        Dim arrNew As Variant
        arrNew = dictUpload(key) ' 6 elements
        
        If Not dictExisting.Exists(key) Then
            ' => NEW
            AddNewAccount arrNew
        Else
            ' => Possibly CHANGED
            If allowOverwrites Then
                Dim arrOld As Variant
                arrOld = dictExisting(key)
                If AreDifferent(arrOld, arrNew) Then
                    OverwriteAccount key, arrOld, arrNew
                End If
            End If
        End If
    Next key
    
Cleanup:
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Exit Sub
    
ErrHandler:
    MsgBox "Error in ImportAccounts: " & Err.Description, vbCritical
    Resume Cleanup
End Sub

' ==================================================
' BUILD DICTIONARIES
' ==================================================

' Build a dictionary from "Account Data"
'   Key = "RelNum|MasterID|AcctNum"
'   Value = 1D array (6 elements)
Private Sub BuildDictFromAccountData(dict As Object)
    Dim ws As Worksheet
    Dim lastRow As Long, i As Long
    
    Set ws = ThisWorkbook.Worksheets(SH_ACCOUNT_DATA)
    lastRow = ws.Cells(ws.Rows.Count, cRelNum).End(xlUp).Row
    
    Dim key As String
    For i = 2 To lastRow
        Dim relNum As String, relName As String
        Dim masterID As String, masterName As String
        Dim acctNum As String, acctName As String
        
        ' Wrap each cell read in CStr() to avoid type mismatch
        relNum = CleanString(ws.Cells(i, cRelNum).Value)
        relName = CleanString(ws.Cells(i, cRelName).Value)
        masterID = CleanString(ws.Cells(i, cMasterID).Value)
        masterName = CleanString(ws.Cells(i, cMasterName).Value)
        acctNum = CleanString(ws.Cells(i, cAcctNum).Value)
        acctName = CleanString(ws.Cells(i, cAcctName).Value)
        
        key = relNum & "|" & masterID & "|" & acctNum
        
        Dim arr(0 To 5) As String
        arr(0) = relNum
        arr(1) = relName
        arr(2) = masterID
        arr(3) = masterName
        arr(4) = acctNum
        arr(5) = acctName
        
        dict(key) = arr
    Next i
End Sub

' Build a dictionary from the external file (variable column order)
'   Key = "RelNum|MasterID|AcctNum"
'   Value = array(6 elements)
Private Sub BuildDictFromExternalFile(ByVal filePath As String, dict As Object)
    Dim wb As Workbook, ws As Worksheet
    Dim lastRow As Long, lastCol As Long
    Dim colMap As Object
    
    On Error GoTo ErrHandler
    Set wb = Workbooks.Open(Filename:=filePath, ReadOnly:=True)
    Set ws = wb.Sheets(1)
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    
    ' Map the columns by header name
    Set colMap = CreateObject("Scripting.Dictionary")
    colMap.CompareMode = vbTextCompare
    
    BuildColumnMap ws, lastCol, colMap  ' Fills colMap("Relationship#")=1, etc.
    
    Dim i As Long
    For i = 2 To lastRow
        Dim relNum As String, relName As String
        Dim masterID As String, masterName As String
        Dim acctNum As String, acctName As String
        
        relNum = CleanString(ws.Cells(i, colMap("Relationship#")).Value)
        relName = CleanString(ws.Cells(i, colMap("Relationship Name")).Value)
        masterID = CleanString(ws.Cells(i, colMap("Master Account ID")).Value)
        masterName = CleanString(ws.Cells(i, colMap("Master Account Name")).Value)
        acctNum = CleanString(ws.Cells(i, colMap("Account#")).Value)
        acctName = CleanString(ws.Cells(i, colMap("Acct Name")).Value)
        
        Dim key As String
        key = relNum & "|" & masterID & "|" & acctNum
        
        Dim arr(0 To 5) As String
        arr(0) = relNum
        arr(1) = relName
        arr(2) = masterID
        arr(3) = masterName
        arr(4) = acctNum
        arr(5) = acctName
        
        dict(key) = arr
    Next i
    
    wb.Close SaveChanges:=False
    Exit Sub
ErrHandler:
    MsgBox "Error reading external file: " & Err.Description, vbCritical
    On Error Resume Next
    wb.Close False
End Sub

' BuildColumnMap: identifies the column numbers for each required header
Private Sub BuildColumnMap(ws As Worksheet, lastCol As Long, dict As Object)
    Dim headerRow As Long: headerRow = 1
    Dim c As Long
    
    Dim foundCount As Long: foundCount = 0
    
    ' Load the array of required headers
    SetupRequiredHeaders
    Dim needed As Object
    Set needed = CreateObject("Scripting.Dictionary")
    needed.CompareMode = vbTextCompare
    
    Dim h As Variant
    For Each h In requiredHeaders
        needed(h) = True
    Next h
    
    ' Scan row 1 for these headers
    For c = 1 To lastCol
        Dim hdrVal As String
        hdrVal = CleanString(ws.Cells(headerRow, c).Value)
        
        If needed.Exists(hdrVal) Then
            dict(hdrVal) = c  ' store column index
            foundCount = foundCount + 1
        End If
    Next c
    
    If foundCount < UBound(requiredHeaders) + 1 Then
        Err.Raise vbObjectError + 555, , "Missing one or more required column headers."
    End If
End Sub

' ==================================================
' COMPARISON & UPDATES
' ==================================================

' AreDifferent: compares 2 arrays(6 strings) for any mismatch
Private Function AreDifferent(oldArr() As String, newArr() As String) As Boolean
    Dim i As Long
    For i = 0 To 5
        If oldArr(i) <> newArr(i) Then
            AreDifferent = True
            Exit Function
        End If
    Next i
    AreDifferent = False
End Function

' AddNewAccount: appends a new row in "Account Data" and highlights green
Private Sub AddNewAccount(arrNew() As String)
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets(SH_ACCOUNT_DATA)
    
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, cRelNum).End(xlUp).Row + 1
    
    ws.Cells(lastRow, cRelNum).Value = arrNew(0)
    ws.Cells(lastRow, cRelName).Value = arrNew(1)
    ws.Cells(lastRow, cMasterID).Value = arrNew(2)
    ws.Cells(lastRow, cMasterName).Value = arrNew(3)
    ws.Cells(lastRow, cAcctNum).Value = arrNew(4)
    ws.Cells(lastRow, cAcctName).Value = arrNew(5)
    
    ' Highlight row in green
    ws.Range(ws.Cells(lastRow, cRelNum), _
             ws.Cells(lastRow, cAcctName)).Interior.Color = RGB(144, 238, 144)
    
    ' Log in "Change Log"
    LogChange "NEW", "Key=" & arrNew(0) & "|" & arrNew(2) & "|" & arrNew(4), "", ""
End Sub

' OverwriteAccount: modifies an existing row (found by key), highlight in yellow
Private Sub OverwriteAccount(ByVal key As String, _
                             oldArr() As String, _
                             newArr() As String)
    
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets(SH_ACCOUNT_DATA)
    
    ' Find the existing row
    Dim rowFound As Long
    rowFound = FindAccountRow(oldArr(0), oldArr(2), oldArr(4))
    If rowFound = 0 Then Exit Sub ' doesn't exist => no overwrite possible
    
    ' Save old vs new for logging
    Dim oldVal As String
    oldVal = oldArr(1) & " / " & oldArr(3) & " / " & oldArr(5)
    Dim newVal As String
    newVal = newArr(1) & " / " & newArr(3) & " / " & newArr(5)
    
    ' Update
    ws.Cells(rowFound, cRelName).Value = newArr(1)
    ws.Cells(rowFound, cMasterID).Value = newArr(2)
    ws.Cells(rowFound, cMasterName).Value = newArr(3)
    ws.Cells(rowFound, cAcctName).Value = newArr(5)
    
    ' Highlight in yellow
    ws.Range(ws.Cells(rowFound, cRelNum), _
             ws.Cells(rowFound, cAcctName)).Interior.Color = RGB(255, 255, 153)
    
    LogChange "CHANGED", "Key=" & key, oldVal, newVal
End Sub

' FindAccountRow: looks up the row in "Account Data" matching (Rel#, MasterID, AcctNum)
Private Function FindAccountRow(ByVal relNum As String, _
                                ByVal masterID As String, _
                                ByVal acctNum As String) As Long
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets(SH_ACCOUNT_DATA)
    
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, cRelNum).End(xlUp).Row
    
    Dim r As Long
    For r = 2 To lastRow
        If (CleanString(ws.Cells(r, cRelNum).Value) = relNum) _
            And (CleanString(ws.Cells(r, cMasterID).Value) = masterID) _
            And (CleanString(ws.Cells(r, cAcctNum).Value) = acctNum) Then
                FindAccountRow = r
                Exit Function
        End If
    Next r
    FindAccountRow = 0
End Function

' ==================================================
' LOGGING & HELPER ROUTINES
' ==================================================

' LogChange: writes an entry in "Change Log"
Private Sub LogChange(ByVal actionDesc As String, ByVal affectedData As String, _
                      Optional ByVal oldVal As String = "", _
                      Optional ByVal newVal As String = "")
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets(SH_CHANGE_LOG)
    
    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, chgTimestamp).End(xlUp).Row + 1
    
    ws.Cells(nextRow, chgTimestamp).Value = Now
    ws.Cells(nextRow, chgUser).Value = Environ("USERNAME")
    ws.Cells(nextRow, chgAction).Value = actionDesc
    ws.Cells(nextRow, chgAffectedData).Value = affectedData
    If oldVal <> "" Then ws.Cells(nextRow, chgOldValue).Value = oldVal
    If newVal <> "" Then ws.Cells(nextRow, chgNewValue).Value = newVal
End Sub

' Cleans a variant into a safe string, avoiding type mismatches
Private Function CleanString(ByVal val As Variant) As String
    On Error Resume Next
    If IsError(val) Then
        CleanString = ""
    ElseIf IsEmpty(val) Or IsNull(val) Then
        CleanString = ""
    Else
        CleanString = CStr(val)
    End If
End Function
