'==============================
' Code for frmRelationship
'==============================
Option Explicit

Private Sub UserForm_Initialize()
    LoadAccountList
End Sub

Private Sub LoadAccountList()
    Dim ws As Worksheet, lo As ListObject, lr As ListRow
    Set ws = ThisWorkbook.Sheets("Accounts")
    Set lo = ws.ListObjects("tblAccounts")
    lstAvailableAccounts.Clear
    For Each lr In lo.ListRows
        lstAvailableAccounts.AddItem lr.Range.Cells(1, 3).Value  ' Assumes Account# is column 3
    Next lr
End Sub

Private Sub cmdAdd_Click()
    Dim selIndex As Long
    selIndex = lstAvailableAccounts.ListIndex
    If selIndex <> -1 Then
        lstSelectedAccounts.AddItem lstAvailableAccounts.List(selIndex)
        lstAvailableAccounts.RemoveItem selIndex
    Else
        MsgBox "Please select an account to add.", vbExclamation
    End If
End Sub

Private Sub cmdRemove_Click()
    Dim selIndex As Long
    selIndex = lstSelectedAccounts.ListIndex
    If selIndex <> -1 Then
        lstAvailableAccounts.AddItem lstSelectedAccounts.List(selIndex)
        lstSelectedAccounts.RemoveItem selIndex
    Else
        MsgBox "Please select an account to remove.", vbExclamation
    End If
End Sub

Private Sub cmdSave_Click()
    Dim wsRel As Worksheet: Set wsRel = ThisWorkbook.Sheets("Relationships")
    Dim nextRow As Long, i As Long
    
    If Trim(txtRelationshipName.Value) = "" Then
        MsgBox "Please enter a Relationship name.", vbExclamation
        Exit Sub
    End If
    
    If lstSelectedAccounts.ListCount = 0 Then
        MsgBox "Please add at least one account to the relationship.", vbExclamation
        Exit Sub
    End If
    
    nextRow = wsRel.Cells(wsRel.Rows.Count, 1).End(xlUp).Row + 1
    For i = 0 To lstSelectedAccounts.ListCount - 1
        With wsRel
            .Cells(nextRow, 1).Value = nextRow - 1  ' Simple incremental ID
            .Cells(nextRow, 2).Value = txtRelationshipName.Value
            .Cells(nextRow, 3).Value = ""           ' Parent ID (if needed)
            .Cells(nextRow, 4).Value = lstSelectedAccounts.List(i)
            .Cells(nextRow, 5).Value = GetMasterID(lstSelectedAccounts.List(i))
        End With
        nextRow = nextRow + 1
    Next i
    
    UpdateAccountRelationships txtRelationshipName.Value, lstSelectedAccounts
    
    MsgBox "Relationship saved successfully.", vbInformation
    Unload Me
End Sub

' Helper: Look up Master Account ID for a given Account#
Private Function GetMasterID(acctID As Variant) As Variant
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets("Accounts")
    Dim lo As ListObject: Set lo = ws.ListObjects("tblAccounts")
    Dim rngFound As Range
    Set rngFound = lo.DataBodyRange.Columns(3).Find(What:=acctID, LookIn:=xlValues, LookAt:=xlWhole)
    If Not rngFound Is Nothing Then
        GetMasterID = rngFound.Offset(0, -2).Value  ' Assumes column 1 is Master Account ID
    Else
        GetMasterID = ""
    End If
End Function

' Update the Accounts table with the new Relationship name.
Private Sub UpdateAccountRelationships(relName As String, selList As MSForms.ListBox)
    Dim ws As Worksheet: Set ws = ThisWorkbook.Sheets("Accounts")
    Dim lo As ListObject: Set lo = ws.ListObjects("tblAccounts")
    Dim i As Long, acctID As Variant, lr As ListRow
    For i = 0 To selList.ListCount - 1
        acctID = selList.List(i)
        For Each lr In lo.ListRows
            If lr.Range.Cells(1, 3).Value = acctID Then
                lr.Range.Cells(1, 9).Value = relName  ' Column 9 is "Relationship"
                Exit For
            End If
        Next lr
    Next i
End Sub
